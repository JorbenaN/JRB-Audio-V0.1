# Spike S3: editor rendering benchmark, as a real signed macOS app bundle built
# entirely by CMake (also verifies the app-bundle half of ADR-017).
file(GLOB S3_SOURCES CONFIGURE_DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/Sources/*.swift)

add_executable(JRBS3Bench MACOSX_BUNDLE ${S3_SOURCES})
target_link_libraries(JRBS3Bench PRIVATE jrb_core)
set_target_properties(JRBS3Bench PROPERTIES
    OUTPUT_NAME "JRB S3 Bench"
    MACOSX_BUNDLE_INFO_PLIST ${CMAKE_CURRENT_SOURCE_DIR}/Info.plist.in
    MACOSX_BUNDLE_GUI_IDENTIFIER org.jrbaudio.s3bench
    MACOSX_BUNDLE_BUNDLE_NAME "JRB S3 Bench"
    MACOSX_BUNDLE_SHORT_VERSION_STRING ${PROJECT_VERSION}
    MACOSX_BUNDLE_BUNDLE_VERSION ${PROJECT_VERSION}
    XCODE_ATTRIBUTE_PRODUCT_BUNDLE_IDENTIFIER org.jrbaudio.s3bench
    XCODE_ATTRIBUTE_SWIFT_OBJC_INTEROP_MODE objcxx
    XCODE_ATTRIBUTE_CLANG_CXX_LANGUAGE_STANDARD c++20
    XCODE_ATTRIBUTE_CODE_SIGN_IDENTITY "-"
    XCODE_ATTRIBUTE_CODE_SIGN_STYLE Manual
    XCODE_ATTRIBUTE_ENABLE_HARDENED_RUNTIME YES
    XCODE_ATTRIBUTE_CODE_SIGN_ENTITLEMENTS ${CMAKE_CURRENT_SOURCE_DIR}/S3Bench.entitlements
)
